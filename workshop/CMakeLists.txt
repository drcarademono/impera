# Reuse the engine's equipment names rather than maintaining another authoring table.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../src/vars.c" engine_names_source)
string(FIND "${engine_names_source}" "char* D_17f6" equipment_begin)
string(FIND "${engine_names_source}" "char* D_1856" equipment_end)
math(EXPR equipment_length "${equipment_end}-${equipment_begin}")
string(SUBSTRING "${engine_names_source}" ${equipment_begin} ${equipment_length} equipment_source)
string(REGEX MATCHALL "_STATIC_TEXT\\([^,]+, \"[^\"]*\"\\)" equipment_matches "${equipment_source}")
list(LENGTH equipment_matches equipment_count)
if(NOT equipment_count EQUAL 48)
    message(FATAL_ERROR "Expected the engine's 48 equipment names")
endif()
set(equipment_header "static const char *engineEquipmentNames[] = {\n")
foreach(item IN LISTS equipment_matches)
    string(REGEX REPLACE ".*, (\"[^\"]*\").*" "\\1" item_name "${item}")
    string(APPEND equipment_header "${item_name},\n")
endforeach()
string(APPEND equipment_header "};\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/inventory_names.h" "${equipment_header}")
# Spell inventory labels use the engine's incantations and syllable dictionary.
string(FIND "${engine_names_source}" "char* D_1bfc" syllables_begin)
string(FIND "${engine_names_source}" "char* D_1c30" spells_begin)
string(FIND "${engine_names_source}" "u8 D_1c90" spells_end)
math(EXPR syllables_length "${spells_begin}-${syllables_begin}")
math(EXPR spells_length "${spells_end}-${spells_begin}")
string(SUBSTRING "${engine_names_source}" ${syllables_begin} ${syllables_length} syllables_source)
string(SUBSTRING "${engine_names_source}" ${spells_begin} ${spells_length} spells_source)
string(REPLACE "char* D_1bfc" "static const char *engineSpellSyllables" syllables_source "${syllables_source}")
string(REPLACE "char* D_1c30" "static const char *engineSpellCodes" spells_source "${spells_source}")
string(REGEX REPLACE "_STATIC_TEXT\\([^,]+, (\"[^\"]*\")\\)" "\\1" syllables_source "${syllables_source}")
string(REGEX REPLACE "_STATIC_TEXT\\([^,]+, (\"[^\"]*\")\\)" "\\1" spells_source "${spells_source}")
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/inventory_names.h" "${syllables_source}\n${spells_source}")

# Dungeon order is the engine's final eight location names.
string(FIND "${engine_names_source}" "char* D_1e3a" dungeon_names_begin)
string(FIND "${engine_names_source}" "u8 D_1e8a" dungeon_names_end)
math(EXPR dungeon_names_length "${dungeon_names_end}-${dungeon_names_begin}")
string(SUBSTRING "${engine_names_source}" ${dungeon_names_begin} ${dungeon_names_length} dungeon_names_source)
string(REGEX MATCHALL "_STATIC_TEXT\\([^,]+, \"[^\"]*\"\\)" dungeon_name_matches "${dungeon_names_source}")
list(LENGTH dungeon_name_matches dungeon_name_count)
math(EXPR dungeon_name_start "${dungeon_name_count}-8")
list(SUBLIST dungeon_name_matches ${dungeon_name_start} 8 dungeon_name_matches)
set(dungeon_names_header "static const char *engineDungeonNames[] = {\n")
foreach(item IN LISTS dungeon_name_matches)
    string(REGEX REPLACE ".*, (\"[^\"]*\").*" "\\1" item_name "${item}")
    string(APPEND dungeon_names_header "${item_name},\n")
endforeach()
string(APPEND dungeon_names_header "};\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/dungeon_names.h" "${dungeon_names_header}")
# Sign border shortcuts are the engine's runic expansions, not a duplicate table.
string(FIND "${engine_names_source}" "char* D_3810" sign_begin)
string(FIND "${engine_names_source}" "u8 D_3822" sign_end)
math(EXPR sign_length "${sign_end}-${sign_begin}")
string(SUBSTRING "${engine_names_source}" ${sign_begin} ${sign_length} sign_source)
string(REGEX MATCHALL "_STATIC_TEXT\\([^,]+, \"[^\"]*\"\\)" sign_matches "${sign_source}")
list(LENGTH sign_matches sign_count)
if(NOT sign_count EQUAL 9)
    message(FATAL_ERROR "Expected nine engine sign border shortcuts")
endif()
set(sign_header "static const char *engineSignBorders[] = {\n")
foreach(item IN LISTS sign_matches)
    string(REGEX REPLACE ".*, (\"[^\"]*\").*" "\\1" item_name "${item}")
    string(APPEND sign_header "${item_name},\n")
endforeach()
string(APPEND sign_header "};\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sign_borders.h" "${sign_header}")
# Actor labels are derived from the engine's authoritative table.
string(FIND "${engine_names_source}" "char* D_18b6" actor_begin)
string(FIND "${engine_names_source}" "char* D_1916" actor_end)
math(EXPR actor_length "${actor_end}-${actor_begin}")
string(SUBSTRING "${engine_names_source}" ${actor_begin} ${actor_length} actor_source)
string(REGEX MATCHALL "_STATIC_TEXT\\([^,]+, \"[^\"]*\"\\)" actor_matches "${actor_source}")
list(LENGTH actor_matches actor_count)
if(NOT actor_count EQUAL 48)
    message(FATAL_ERROR "Expected 48 engine actor labels")
endif()
set(actor_header "static const char *engineActorNames[] = {\n")
foreach(item IN LISTS actor_matches)
    string(REGEX REPLACE ".*, (\"[^\"]*\").*" "\\1" actor_name "${item}")
    string(APPEND actor_header "${actor_name},\n")
endforeach()
string(APPEND actor_header "};\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/actor_names.h" "${actor_header}")
string(REGEX MATCH "u32 D_28c6\\[0x20\\] =[^;]+;" npc_properties "${engine_names_source}")
if(NOT npc_properties)
    message(FATAL_ERROR "Expected the engine's NPC property flags")
endif()
string(REPLACE "u32 D_28c6" "static const uint32_t nativeNpcProperties" npc_properties "${npc_properties}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native_npc_properties.h" "${npc_properties}\n")
string(REGEX MATCH "u8 D_13a6\\[8\\] =[^;]+;" shadowlord_rows "${engine_names_source}")
if(NOT shadowlord_rows)
    message(FATAL_ERROR "Expected the engine's Shadowlord entry rows")
endif()
string(REPLACE "u8 D_13a6" "static const uint8_t nativeShadowlordRows" shadowlord_rows "${shadowlord_rows}")
file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/native_npc_properties.h" "${shadowlord_rows}\n")
