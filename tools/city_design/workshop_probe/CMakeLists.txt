cmake_minimum_required(VERSION 3.16)
project(CityDesignWorkshopProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(IMPERA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(Qt6 6.2 REQUIRED COMPONENTS Widgets)
set(IMPERA_WORKSHOP_BUILD "" CACHE PATH "Optional freshly built standalone Workshop directory")
if(IMPERA_WORKSHOP_BUILD)
    # Reuse a current build without modifying the engine or Workshop targets.
    add_library(city_workshop_core STATIC IMPORTED)
    set_target_properties(city_workshop_core PROPERTIES
        IMPORTED_LOCATION "${IMPERA_WORKSHOP_BUILD}/libimpera_workshop_core.a")
    file(GLOB_RECURSE resource_objects
        "${IMPERA_WORKSHOP_BUILD}/CMakeFiles/impera_workshop_core_resources_*.dir/*.o")
    if(NOT resource_objects)
        message(FATAL_ERROR "Build impera_workshop_core and its Qt resources first")
    endif()
    set(core city_workshop_core)
else()
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    add_subdirectory("${IMPERA_ROOT}/workshop" workshop EXCLUDE_FROM_ALL)
    set(core impera_workshop_core)
endif()
add_executable(city_workshop_probe probe.cpp ${resource_objects})
target_include_directories(city_workshop_probe PRIVATE "${IMPERA_ROOT}/workshop" "${IMPERA_ROOT}/src")
target_link_libraries(city_workshop_probe PRIVATE ${core} Qt6::Widgets)
